cmake_minimum_required(VERSION 3.18)
project(log_pipeline VERSION 2.0 LANGUAGES C CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} ${CMAKE_BINARY_DIR}/bin)
endforeach()
include(FetchContent)
find_package(Threads REQUIRED)
FetchContent_Declare(json URL https://github.com/nlohmann/json/archive/refs/tags/v3.11.3.tar.gz)
FetchContent_Declare(httplib URL https://github.com/yhirose/cpp-httplib/archive/refs/tags/v0.18.7.tar.gz SOURCE_SUBDIR unused)
FetchContent_Declare(sha2 URL https://github.com/okdshin/PicoSHA2/archive/161cb3fc4170fa7a3eca9e582cebd27cc4d1fe29.tar.gz SOURCE_SUBDIR unused)
FetchContent_MakeAvailable(json httplib sha2)
if(NOT LOG_SQLITE_DIR)
  FetchContent_Declare(sqlite_source URL https://github.com/SRombauts/SQLiteCpp/archive/refs/tags/3.3.3.tar.gz SOURCE_SUBDIR unused)
  FetchContent_MakeAvailable(sqlite_source)
  set(LOG_SQLITE_DIR ${sqlite_source_SOURCE_DIR}/sqlite3)
endif()
add_library(log_sqlite STATIC ${LOG_SQLITE_DIR}/sqlite3.c)
target_include_directories(log_sqlite PUBLIC ${LOG_SQLITE_DIR})
target_compile_definitions(log_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DEFAULT_WAL_SYNCHRONOUS=2)
target_link_libraries(log_sqlite PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
add_executable(log_queue queue/src/main.cpp queue/src/sqlite_queue.cpp)
add_executable(log_agent agent/src/main.cpp agent/src/log_parser.cpp)
target_include_directories(log_agent PRIVATE agent/include)
foreach(target log_queue log_agent)
  target_include_directories(${target} PRIVATE native/include ${httplib_SOURCE_DIR} ${sha2_SOURCE_DIR})
  target_link_libraries(${target} PRIVATE nlohmann_json::nlohmann_json Threads::Threads)
  if(WIN32)
    target_link_libraries(${target} PRIVATE ws2_32)
    target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  endif()
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
target_link_libraries(log_queue PRIVATE log_sqlite)
